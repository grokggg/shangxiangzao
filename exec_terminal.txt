############## nexus/runtime/executors/terminal/session.py ##############
"""TerminalSession: single-PTY shell with file-based bash-capture.

The shell runs under one PTY so interaction, job control, and capture_pane
(pyte) keep working. A patched bash, built with --wrap=execute_command and
--wrap=readline, writes per-command metadata and stdout/stderr/output files.
Python polls those files instead of depending on shell-mutable state.

Multi-line input is fed raw, not wrapped in `{ ...; }`. Bash executes it as
several commands that share one batch id; run_command() and get_command_info()
aggregate the batch into one CommandInfo.
"""

import asyncio
import logging
import os
import shutil
import time
import uuid

from nexus.runtime.executors.models.terminal import CommandInfo
from nexus.runtime.executors.terminal.builtin import locate_patched_bash
from nexus.runtime.executors.terminal.capture import CaptureReader, CommandRecord
from nexus.runtime.executors.terminal.keys import keys_to_bytes
from nexus.runtime.executors.terminal.pty_io import PtyProcess
from nexus.runtime.executors.terminal.screen import VirtualScreen
from nexus.utils.paths import TOOLS_DIR
from nexus.utils.shell import get_clean_sandbox_env, get_default_pwd


logger = logging.getLogger(__name__)

# A batch's commands register ~0.25ms apart (measured max 0.32ms, independent of
# command runtime — it's just "bash reads next line + writes the file"). After a
# batch's known commands all finish, wait this long and re-check before declaring
# the batch done, to cover the sub-ms gap before the next command registers.
BATCH_SETTLE_SECONDS = 0.05


def _is_empty_submission(command: str) -> bool:
    """Return True for input that Bash will not execute as a command.

    Empty, newline-only, and whitespace-only submissions do not invoke Bash's
    execute_command(), so the capture layer never writes cmd_<id>.json. We
    synthesize the reserved command_id 0 no-op result instead of waiting for a
    capture record that cannot appear.
    """
    return command.strip() == ''


class TerminalSession:
    def __init__(
        self,
        session_id: str | None = None,
        height: int = 40,
        width: int = 120,
    ) -> None:
        self.session_id = session_id or f'nexus-{uuid.uuid4().hex[:8]}'
        self.height = height
        self.width = width
        self.output_dir = str(TOOLS_DIR / 'terminal' / self.session_id)
        os.makedirs(self.output_dir, exist_ok=True)

        self._screen = VirtualScreen(rows=height, cols=width)
        self._capture = CaptureReader(self.output_dir)
        self._pty: PtyProcess | None = None
        self._dead = False

    async def start(self) -> None:
        bash = locate_patched_bash()

        argv = [bash, '--noprofile', '--norc', '-i']

        env = get_clean_sandbox_env()
        # TERM drives interactive programs (REPLs, pagers, colors) and pyte's
        # emulation for capture_pane.
        env['TERM'] = 'xterm-256color'

        # Enable bash-capture for this session. The patched bash already contains
        # the execute_command/readline link-time wrappers; these env vars only
        # turn capture on and select the output directory. No preloaded shared
        # object is used, so child processes do not inherit capture hooks.
        env['NEXUS_CAPTURE'] = '1'
        env['NEXUS_CAPTURE_DIR'] = self.output_dir

        # Per-command stdout/stderr capture is pipe-only. cmd_wrap.c hijacks each
        # foreground command's fd 1/2 away from the session PTY into pipes, so
        # isatty(fd1) is False and tools fall back to predictable non-interactive
        # behavior (for example, ls auto-disables color and git skips its pager).
        # We intentionally keep only one PTY for the shell; adding per-stream
        # capture PTYs would split terminal control between stdin and output.

        cwd = get_default_pwd()
        env['PWD'] = cwd

        self._pty = await PtyProcess.spawn(
            argv=argv,
            env=env,
            rows=self.height,
            cols=self.width,
            on_data=self._on_data,
            on_exit=self._on_exit,
        )
        try:
            # Readiness does not require a shell command handshake. The mechanism
            # is compiled into bash and activated by the env vars above. The first
            # readline hook writes status.json, which is the ready signal.
            loop = asyncio.get_running_loop()
            deadline = loop.time() + 5.0
            ready = False
            while loop.time() < deadline and not self._dead:
                if await self._capture.read_status() is not None:
                    ready = True
                    break
                await asyncio.sleep(0.05)
            if not ready:
                raise RuntimeError(f'Session {self.session_id}: capture mechanism did not initialize (no status.json)')
        except BaseException:
            self._dead = True
            await self.close()
            raise

    async def close(self) -> None:
        if self._pty is not None:
            self._pty.killpg()
            self._pty.close()
            self._pty = None
        self._screen.close()
        shutil.rmtree(self.output_dir, ignore_errors=True)

    def _on_exit(self) -> None:
        self._dead = True

    def _on_data(self, data: bytes) -> None:
        self._screen.feed(data)

    def _ensure_alive(self) -> PtyProcess:
        if self._pty is None or self._dead:
            raise RuntimeError(f'Session {self.session_id} is not alive')
        return self._pty

    async def start_command(self, command: str) -> int:
        pty = self._ensure_alive()
        if _is_empty_submission(command):
            return 0
        status = await self._capture.read_status()
        if status is not None and status.status == 'command_running':
            raise RuntimeError('A command is already running')
        prev_id = status.command_id if status is not None else 0
        # command_id increments monotonically from the shared counter, so the
        # first command of this submission is deterministically prev_id + 1.
        # (Multi-line input runs as several commands; we return the FIRST id and
        # aggregate the batch from it — never the heuristic "whatever id status
        # shows", which could be the LAST id for a fast multi-line.)
        first_id = prev_id + 1

        # Feed the raw command (no { } wrapping — keeps capture_pane showing the
        # agent's original text). A trailing newline submits the final line.
        await pty.write(command.encode('utf-8') + b'\n')

        loop = asyncio.get_running_loop()
        deadline = loop.time() + 5.0
        while loop.time() < deadline and not self._dead:
            batch = await self._capture.read_batch(first_id)
            if batch.records:  # first command registered (cmd_<first_id>.json exists)
                return first_id
            await asyncio.sleep(0.02)
        if self._dead:
            raise RuntimeError(f'Session {self.session_id} died before command registered')
        raise RuntimeError('Command failed to start (capture layer did not register it)')

    async def run_command(self, command: str, timeout: float | None = None) -> CommandInfo:
        loop = asyncio.get_running_loop()
        deadline = None if timeout is None else loop.time() + timeout

        first_id = await self.start_command(command)
        if first_id == 0:
            return self._empty_command_info()

        while True:
            if self._dead:
                raise RuntimeError(f'Session {self.session_id} died')
            batch = await self._capture.read_batch(first_id)
            if batch.records and batch.all_finished:
                # Known commands all done — settle to confirm no new command in
                # this batch is still registering, then aggregate.
                await asyncio.sleep(BATCH_SETTLE_SECONDS)
                batch2 = await self._capture.read_batch(first_id)
                if len(batch2.records) == len(batch.records):
                    return self._aggregate(batch2.records)
                # A new command appeared; keep polling.
            if deadline is not None and loop.time() >= deadline:
                # Match the online implementation's timeout semantics: interrupt
                # the foreground command, then remove every remaining descendant
                # of the session shell (including background jobs). Keep the shell
                # itself alive so the terminal session remains reusable.
                termination_status = 'failed'
                termination_error = None
                try:
                    await self.terminate_session_processes()
                    termination_status = 'succeeded'
                except Exception as exc:  # noqa: BLE001
                    termination_error = str(exc)
                    logger.exception('Failed to terminate processes for command %s', first_id)

                # Briefly wait for the capture layer to finalize any open command
                # files. A mid-batch timeout does not return finished output.
                fin = loop.time() + 2.5
                while loop.time() < fin and not self._dead:
                    batch = await self._capture.read_batch(first_id)
                    if batch.records and batch.all_finished:
                        break
                    await asyncio.sleep(0.02)
                error_msg = (
                    f'Command {first_id} timed out after {timeout} seconds. Process termination {termination_status}'
                )
                if termination_error:
                    error_msg += f': {termination_error}'
                raise TimeoutError(error_msg)
            await asyncio.sleep(0.05)

    @staticmethod
    def _empty_command_info() -> CommandInfo:
        """Synthesized result for an empty submission (reserved command_id 0):
        a no-op success that never touched the shell, so the start poll isn't
        stalled waiting for a command the capture layer never registered."""
        now = time.time()
        return CommandInfo(
            command_id=0,
            command='',
            exit_code=0,
            start_time=now,
            end_time=now,
            stdout='',
            stderr='',
            output='',
        )

    @staticmethod
    def _aggregate(records: list[CommandRecord]) -> CommandInfo:
        """Aggregate a batch's command records (ascending by id, non-empty) into
        one CommandInfo: first id, joined command/stdout/stderr/output, last
        record's exit code. If any record is unfinished, end_time/exit_code are
        None so the whole batch reads as still-running (codeact keys off end_time)."""
        all_finished = all(record.end_time is not None for record in records)
        first = records[0]
        last = records[-1]
        command = '\n'.join(record.command or '' for record in records) or None
        return CommandInfo(
            command_id=first.command_id,
            command=command,
            exit_code=last.exit_code if all_finished else None,
            start_time=first.start_time,
            end_time=last.end_time if all_finished else None,
            stdout=''.join(record.stdout for record in records),
            stderr=''.join(record.stderr for record in records),
            output=''.join(record.output for record in records),
        )

    async def get_command_info(self, command_id: int) -> CommandInfo | None:
        if command_id == 0:
            return self._empty_command_info()
        batch = await self._capture.read_batch(command_id)
        if not batch.records:
            return None
        if not batch.all_finished:
            # Some command in the batch is still running — fast path, no wait.
            return self._aggregate(batch.records)  # end_time/exit_code None -> in progress
        # Known commands all done — settle to confirm the batch isn't still
        # growing, then aggregate. Stateless: each call re-reads; only pays the
        # window when the batch looks complete.
        await asyncio.sleep(BATCH_SETTLE_SECONDS)
        batch2 = await self._capture.read_batch(command_id)
        # batch2 is normally a superset of batch (append-only files); fall back
        # to the previous records if a concurrent close() emptied the dir.
        return self._aggregate(batch2.records or batch.records)

    async def send_keys(self, keys: str | list[str]) -> bool:
        pty = self._ensure_alive()
        try:
            await pty.write(keys_to_bytes(keys))
            return True
        except OSError:
            return False

    async def capture_pane(self, capture_entire: bool = False) -> str:
        return await self._screen.display(entire=capture_entire)

    async def resize(self, height: int, width: int) -> bool:
        pty = self._ensure_alive()
        try:
            pty.resize(height, width)
            await self._screen.resize(height, width)
            self.height = height
            self.width = width
            return True
        except OSError:
            return False

    async def terminate_session_processes(self) -> None:
        pty = self._ensure_alive()
        if not await self.send_keys('C-c'):
            logger.warning('Failed to send Ctrl+C to session %s', self.session_id)
        await asyncio.sleep(1)
        await pty.terminate_descendants()
