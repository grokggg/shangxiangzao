#!/usr/bin/env bash
# dind-runc — runc wrapper for the inner dockerd inside the DinD sandbox.
#
# At `create` it rewrites the OCI spec:
#   1) bind-mounts the merged CA bundle (public roots + Squid CA) and the
#      squid-only CA into fixed in-container paths;
#   2) injects proxy / mirror / CA env into .process.env (respecting user-set
#      values via `docker run -e`);
#   3) when the image ships a shell, appends a `startContainer` OCI hook that
#      installs the squid CA into the container's own trust stores
#      (system store / certifi / java) — same effect as suanlicluster.py,
#      executed in the container mount ns after pivot_root so it never touches
#      the host.
# Every other runc subcommand is passed through untouched. Any injection
# failure is swallowed so it can never abort container creation.
#
# Runtime state files produced by docker-bootstrap.sh:
#   /var/run/dind-ca-bundle.crt  — public roots + Squid CA (CA env target src)
#   /var/run/dind-inject.env     — proxy + mirror env whitelist snapshot
#   /usr/local/share/ca-certificates/squid-proxy-ca.crt — squid-only CA (host)
#
# This script is packaged by the nix flake into
# /nix/runtime/nix-env/bin/dind-runc — it does NOT need runtime code
# generation. The real runc is resolved from non-wrapper nix store entries to
# avoid recursion after PATH and moby libexec/docker/runc are repointed here;
# jq still uses PATH with a fixed nix-env fallback.
set -euo pipefail

# Resolve the real runc binary. dind-runc is installed as "runc" in both the
# nix profile and moby's libexec/docker/runc, so we must not use
# `command -v runc` — it may resolve back to this wrapper. Instead, select a
# non-wrapper runc from the nix store. When BuildKit reaches us through moby's
# libexec/docker/runc path, prefer docker-runc; otherwise prefer standalone runc
# and fall back to docker-runc if standalone runc is not present in the closure.
RUNC_BIN=""
case "$0" in
    *moby*libexec/docker/runc*|*libexec/docker/runc*)
        for c in /nix/store/*-docker-runc-*/bin/runc; do
            [[ -x "$c" ]] && { RUNC_BIN="$c"; break; }
        done
        ;;
    *)
        for c in /nix/store/*-runc-*/bin/runc; do
            case "$c" in *docker-runc*|*dind-runc*) continue ;; esac
            [[ -x "$c" ]] && { RUNC_BIN="$c"; break; }
        done
        if [[ ! -x "$RUNC_BIN" ]]; then
            for c in /nix/store/*-docker-runc-*/bin/runc; do
                [[ -x "$c" ]] && { RUNC_BIN="$c"; break; }
            done
        fi
        ;;
esac
if [[ ! -x "$RUNC_BIN" ]]; then
    echo "dind-runc: FATAL: cannot locate real runc binary (called as $0)" >&2
    exit 1
fi

#!/usr/bin/env bash

set -euo pipefail

LOG=/var/log/dockerd-bootstrap.log
# /run must be created before /var/run: in some images /var/run is a symlink to
# a missing /run, and `mkdir -p /var/run` then fails on the dangling symlink.
mkdir -p /var/log /run /var/run

if [ -S /var/run/docker.sock ] && docker info >/dev/null 2>&1; then
    echo "dockerd already running"
    exit 0
fi

if ! iptables --version 2>/dev/null | grep -q legacy; then
    echo "FATAL: iptables must use the legacy backend for Docker-in-Docker under gVisor" >&2
    iptables --version 2>&1 | head -1 >&2 || true
    exit 1
fi

dev="$(ip route show default | awk '/default/ {print $5; exit}')"
if [[ -z "$dev" ]]; then
    echo "FATAL: cannot determine default network device for Docker-in-Docker SNAT" >&2
    exit 1
fi

addr="$(ip -4 addr show dev "$dev" | awk '/inet / {print $2; exit}' | cut -d/ -f1)"
if [[ -z "$addr" ]]; then
    echo "FATAL: cannot determine IPv4 address for device $dev" >&2
    exit 1
fi

echo 1 > /proc/sys/net/ipv4/ip_forward
iptables -t nat -A POSTROUTING -o "$dev" -j SNAT --to-source "$addr" -p tcp 2>>"$LOG" || true
iptables -t nat -A POSTROUTING -o "$dev" -j SNAT --to-source "$addr" -p udp 2>>"$LOG" || true

# Snapshot the outer sandbox's env whitelist for dind-runc to inject into inner
# containers. MUST run before the `export SSL_CERT_FILE` below — otherwise the
# snapshot would capture docker-bootstrap.sh's dockerd-specific CA path
# (/var/run/dind-ca-bundle.crt) instead of the original outer value
# (/etc/ssl/certs/ca-certificates.crt), and inner containers would get a
# SSL_CERT_FILE pointing to a file that doesn't exist inside them.
INJECT_ENV=/var/run/dind-inject.env
: >"$INJECT_ENV"
for k in \
    HTTP_PROXY HTTPS_PROXY FTP_PROXY NO_PROXY \
    http_proxy https_proxy ftp_proxy no_proxy \
    PIP_INDEX_URL PIP_EXTRA_INDEX_URL UV_DEFAULT_INDEX UV_INDEX_URL \
    UV_NATIVE_TLS CARGO_HTTP_MULTIPLEXING GRADLE_OPTS \
    SSL_CERT_FILE REQUESTS_CA_BUNDLE NODE_EXTRA_CA_CERTS; do
    v="${!k:-}"; [[ -n "$v" ]] && printf '%s=%s\n' "$k" "$v" >>"$INJECT_ENV"
done

# Outbound traffic goes through a squid proxy that does TLS interception, so the
# registry cert dockerd sees is re-signed by a custom squid CA — without it
# dockerd fails with "x509: certificate signed by unknown authority". The task
# image rootfs (e.g. ubuntu:24.04) may also lack ca-certificates entirely.
# We therefore concatenate a public-root bundle (best effort) with the custom

# registry-agnostic: it applies to every registry dockerd talks to.

    /nix/runtime/nix-env/etc/ssl/certs/ca-bundle.crt \
    /etc/ssl/certs/ca-certificates.crt \
    /etc/ssl/certs/ca-bundle.crt \
    /nix/store/*-nss-cacert-*/etc/ssl

def _resolve_trim_threshold(args: argparse.Namespace) -> tuple[int, int]:
    """Build the ``(head, tail)`` trim_threshold tuple from CLI args.

    ``--log-trim N`` sets both head and tail to *N*; the per-side flags then
    override either end. Unset flags fall back to 200.
    """
    both = getattr(args, 'log_trim', None)
    default = 200 if both is None else int(both)
    head_raw = getattr(args, 'log_trim_head', None)
    tail_raw = getattr(args, 'log_trim_tail', None)
    head = default if head_raw is None else int(head_raw)
    tail = default if tail_raw is None else int(tail_raw)
    return max(head, 0), max(tail, 0)

            return 1

            run_input = compile_single(

    if not base_path or base_path == '/':

## runtime-readiness.py (完整)
#!/usr/bin/python3
"""Require live service health, never a snapshot marker."""
import argparse
import json
import os
import socket
import sys
import time
import urllib.error
import urllib.request


def probe(host, port):
    try:
        with socket.create_connection((host, port), timeout=1):
            return True
    except OSError:
        return False


class NoRedirect(urllib.request.HTTPRedirectHandler):
    def redirect_request(self, req, fp, code, msg, headers, newurl):
        return None


def probe_health(port, path, expected_status):
    # Always contact this container directly, without inherited HTTP proxies.
    opener = urllib.request.build_opener(urllib.request.ProxyHandler({}), NoRedirect())
    try:
        with opener.open(f'http://127.0.0.1:{port}{path}', timeout=1) as response:
            if response.status != 200:
                return False
            data = response.read(65537)
            if len(data) > 65536:
                return False
            body = json.loads(data)
            return isinstance(body, dict) and body.get('status') == expected_status
    except (OSError, ValueError, urllib.error.URLError):
        return False


def check_ready(port_probe=probe, health_probe=probe_health, *, dependencies_only=False):
    ports = [int(os.environ.get('NEXUS_ENVD_PORT', '49983')),
             int(os.environ.get('SANDBOXD_PORT', '20001')),
             int(os.environ.get('NEXUS_RUNTIME_PORT', '32000'))]
    if any(not 1 <= port <= 65535 for port in ports) or len(set(ports)) != 3:
        raise ValueError('service ports must be distinct and between 1 and 65535')
    results = [port_probe('127.0.0.1', ports[0]),
               health_probe(ports[2], '/health', 'ok')]
    # sandboxd invokes this mode from its own HTTP handler. Probing that handler
    # again would recurse; the caller's HTTP response proves sandboxd is serving.
    if not dependencies_only:
        results.append(health_probe(ports[1], '/api/v1/runtime/health', 'READY'))
    return all(results)


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--wait', action='store_true')
    parser.add_argument('--supervisor-pid', type=int)
    parser.add_argument('--dependencies-only', action='store_true')
    args = parser.parse_args()
    timeout = float(os.environ.get('RUNTIME_START_TIMEOUT_SECONDS', '120')) if args.wait else 0
    deadline = time.monotonic() + timeout
    while True:
        if args.supervisor_pid:
            try:
                os.kill(args.supervisor_pid, 0)
            except ProcessLookupError:
                print('Supervisor exited before all service ports were listening', file=sys.stderr)
                return 1
        if check_ready(dependencies_only=args.dependencies_only):
            return 0
        if time.monotonic() >= deadline:
            print('envd listener, sandboxd READY and Nexus health ok are required', file=sys.stderr)
            return 1
        time.sleep(0.5)


if __name__ == '__main__':
    sys.exit(main())

    os.execv(args[0], args)

    main()

## nexus-userns.py
#!/usr/bin/python3
"""Map ordinary container users into an isolated user namespace before bwrap."""
import ctypes
import os
import pathlib
import signal
import sys


def main():
    ready_read, ready_write = os.pipe()
    release_read, release_write = os.pipe()
    pid = os.fork()
    if pid == 0:
        os.close(ready_read)
        os.close(release_write)
        libc = ctypes.CDLL(None, use_errno=True)
        if libc.unshare(0x10000000) != 0:  # CLONE_NEWUSER
            os._exit(1)
        os.write(ready_write, b'1')
        os.close(ready_write)
        if os.read(release_read, 1) != b'1':
            os._exit(1)
        os.close(release_read)
        os.execv(sys.argv[1], sys.argv[1:])
    os.close(ready_write)
    os.close(release_read)
    try:
        if os.read(ready_read, 1) != b'1':
            raise RuntimeError('cannot create Nexus user namespace')
        # The trusted parent writes the maps while it still has capabilities in
        # the container namespace. The child cannot modify these maps afterwards.
        for name in ('uid_map', 'gid_map'):
            pathlib.Path(f'/proc/{pid}/{name}').write_text('0 0 65536\n')
        os.write(release_write, b'1')
    except BaseException:
        os.kill(pid, signal.SIGKILL)
        os.waitpid(pid, 0)
        raise
    finally:
        os.close(ready_read)
        os.close(release_write)
    _, status = os.waitpid(pid, 0)
    return os.waitstatus_to_exitcode(status)


if __name__ == '__main__':
    sys.exit(main())
